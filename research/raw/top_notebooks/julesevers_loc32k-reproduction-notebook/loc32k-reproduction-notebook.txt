"""Build gold localization annotations for the Gemma-4 Developer Agent tasks.

For every task we read the repository *at base_commit* straight from git objects
(no checkout), parse the gold `patch`, and map every changed line back to the
innermost enclosing function/class of the pre-patch file (Python AST).

Output: data/gold.jsonl  (one record per task)
"""
import ast, json, re, subprocess, sys, functools
from collections import defaultdict
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
REPO_DIR = {'fastapi/fastapi': 'fastapi_fastapi', 'Textualize/rich': 'Textualize_rich',
            'psf/requests': 'psf_requests', 'encode/httpx': 'encode_httpx'}


def git(repo, *args):
    return subprocess.run(['git', '-C', str(ROOT / 'repos' / REPO_DIR[repo]), *args],
                          capture_output=True, check=True).stdout


@functools.lru_cache(maxsize=None)
def ls_tree(repo, commit):
    out = git(repo, 'ls-tree', '-r', '-l', commit).decode()
    files = {}
    for line in out.splitlines():
        meta, path = line.split('\t', 1)
        _mode, typ, sha, size = meta.split()
        if typ == 'blob':
            files[path] = (sha, int(size) if size != '-' else 0)
    return files


@functools.lru_cache(maxsize=200000)
def blob(repo, sha):
    return git(repo, 'cat-file', 'blob', sha).decode('utf-8', 'replace')


def is_test_path(p):
    parts = p.split('/')
    name = parts[-1]
    return (name.startswith('test_') or name.endswith('_test.py') or name == 'conftest.py'
            or any(x in ('tests', 'test', 'testing') for x in parts[:-1]))


HUNK = re.compile(r'^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@')


def parse_patch(patch):
    """Return {path: {'new_file','deleted','old_lines','added','removed'}}. Hunk-length aware."""
    files = {}
    cur = None
    old_ln = None
    o_left = n_left = 0
    pending_src = None
    for line in patch.splitlines():
        if o_left > 0 or n_left > 0:
            if line.startswith('-'):
                cur['old_lines'].add(old_ln); cur['removed'] += 1; old_ln += 1; o_left -= 1
            elif line.startswith('+'):
                cur['old_lines'].add(max(old_ln - 1, 1)); cur['added'] += 1; n_left -= 1
            elif line.startswith('\\'):
                pass
            else:
                old_ln += 1; o_left -= 1; n_left -= 1
            continue
        if line.startswith('--- '):
            pending_src = line[4:].split('\t')[0]
            continue
        if line.startswith('+++ '):
            dst = line[4:].split('\t')[0]
            path = dst[2:] if dst.startswith('b/') else (pending_src[2:] if dst == '/dev/null' else dst)
            cur = files.setdefault(path, {'new_file': pending_src == '/dev/null', 'deleted': dst == '/dev/null',
                                          'old_lines': set(), 'added': 0, 'removed': 0})
            continue
        m = HUNK.match(line)
        if m and cur is not None:
            old_ln = int(m.group(1))
            o_left = int(m.group(2)) if m.group(2) is not None else 1
            n_left = int(m.group(4)) if m.group(4) is not None else 1
            if old_ln == 0 and o_left == 0:
                cur['new_file'] = True
                old_ln = 1
            continue
    return files


def def_spans(src):
    """List of (qualname, kind, start, end) for every def/class (nested qualnames)."""
    try:
        tree = ast.parse(src)
    except SyntaxError:
        return None
    spans = []

    def visit(node, prefix):
        for ch in ast.iter_child_nodes(node):
            if isinstance(ch, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                q = f'{prefix}.{ch.name}' if prefix else ch.name
                start = min([ch.lineno] + [d.lineno for d in ch.decorator_list])
                spans.append((q, 'class' if isinstance(ch, ast.ClassDef) else 'func', start, ch.end_lineno))
                visit(ch, q)
            else:
                visit(ch, prefix)
    visit(tree, '')
    return spans


def innermost(spans, line):
    best = None
    for q, k, s, e in spans:
        if s <= line <= e and (best is None or (e - s) < (best[3] - best[2])):
            best = (q, k, s, e)
    return best


def main():
    rows = [json.loads(l) for l in open(ROOT / 'data' / 'tasks.jsonl')]
    out = open(ROOT / 'data' / 'gold.jsonl', 'w')
    for r in rows:
        repo, commit = r['repo'], r['base_commit']
        tree = ls_tree(repo, commit)
        pf = parse_patch(r['patch'])
        tpf = parse_patch(r['test_patch'])
        rec = {'instance_id': r['instance_id'], 'repo': repo, 'base_commit': commit, 'files': []}
        for path, info in pf.items():
            ent = {'path': path, 'ext': Path(path).suffix, 'new_file': info['new_file'] or path not in tree,
                   'deleted': info['deleted'], 'added': info['added'], 'removed': info['removed'],
                   'is_test': is_test_path(path), 'units': []}
            if not ent['new_file'] and path.endswith('.py') and path in tree:
                src = blob(repo, tree[path][0])
                spans = def_spans(src)
                ent['n_lines'] = src.count('\n') + 1
                if spans is not None:
                    units = set()
                    for ln in sorted(info['old_lines']):
                        b = innermost(spans, ln)
                        units.add(b[0] if b else '<module>')
                    ent['units'] = sorted(units)
                ent['changed_old_lines'] = sorted(info['old_lines'])
            rec['files'].append(ent)
        rec['test_files'] = sorted(tpf.keys())
        rec['test_new_files'] = sorted(p for p, i in tpf.items() if i['new_file'] or p not in tree)
        out.write(json.dumps(rec) + '\n')
    out.close()
    print('wrote', len(rows))


if __name__ == '__main__':
    main()

"""Training-free localization methods + evaluation under token budgets.

Every method sees exactly what the agent sees at inference time inside Container A:
the problem statement and the repository snapshot at base_commit (including git
history *before* base_commit). Nothing from `patch` / `test_patch` is used for ranking.
"""
import ast, json, math, re, subprocess, functools, sys
from collections import defaultdict, Counter
from pathlib import Path

import numpy as np
import sentencepiece as spm
from rank_bm25 import BM25Okapi

sys.path.insert(0, str(Path(__file__).parent))
from annotate import ROOT, REPO_DIR, ls_tree, blob, is_test_path, def_spans, git

SP = spm.SentencePieceProcessor(model_file=str(ROOT / 'data' / 'gemma3_cleaned_262144_v2.spiece.model'))


@functools.lru_cache(maxsize=None)
def ntok(text):
    return len(SP.encode(text))


# ---------------------------------------------------------------- tokenization
STOP = set('''a an the of to in on for and or is are be by with as at from this that it its
not no if else elif return def class self cls import none true false pass raise try except
finally while yield lambda async await we our you your can should would will just also use
used using add fix fixes fixed support pr pull request issue issues https http github com www
when then than into out so do does there here which what new make one all any but have has

type value'''.split())

CAMEL = re.compile(r'[A-Z]+(?=[A-Z][a-z])|[A-Z]?[a-z]+|[A-Z]+|\d+')


def split_ident(tok):
    parts = []
    for p in re.split(r'[_\W]+', tok):
        if p:
            parts += [x.lower() for x in CAMEL.findall(p)]
    return parts


def text_tokens(text, keep_full=True):
    out = []
    for raw in re.findall(r'[A-Za-z_][A-Za-z0-9_]*', text):
        low = raw.lower()
        if keep_full and ('_' in raw or re.search(r'[a-z][A-Z]', raw)) and low not in STOP:
            out.append(low)  # keep the compound identifier itself
        for p in split_ident(raw):
            if len(p) > 1 and p not in STOP:
                out.append(p)
    return out


def clean_statement(s):
    s = re.sub(r'<!--.*?-->', ' ', s, flags=re.S)  # PR template comments
    s = re.sub(r'https?://\S+', ' ', s)
    return s


IDENT_PATTERNS = [
    r'`([^`\n]{2,80})`',                       # backticked code
    r'\b([A-Za-z_]\w*(?:\.[A-Za-z_]\w*)+)\b',  # dotted paths
    r'\b([A-Z][a-z0-9]+(?:[A-Z][a-z0-9]*)+)\b',  # CamelCase
    r'\b([a-z0-9]+(?:_[a-z0-9]+)+)\b',          # snake_case
    r'\b(\w+\.py)\b',                          # file names
]


def extract_identifiers(s):
    found = set()
    for pat in IDENT_PATTERNS:
        for m in re.findall(pat, s):
            for piece in re.findall(r'[A-Za-z_][A-Za-z0-9_]*', m):
                if len(piece) > 2 and piece.lower() not in STOP:
                    found.add(piece)
    return found


AUX_DIRS = {'docs', 'doc', 'docs_src', 'examples', 'example', 'benchmarks', 'benchmark', 'scripts', 'tools'}


def scoped(idx, rank):
    """Soft scoping: library files first (in rank order), then auxiliary files (in rank order)."""
    return [f for f in rank if f not in idx.aux] + [f for f in rank if f in idx.aux]


# ---------------------------------------------------------------- repository index
class RepoIndex:
    def __init__(self, repo, commit, include_docs_src=True):
        self.repo, self.commit = repo, commit
        tree = ls_tree(repo, commit)
        self.files = sorted(p for p in tree if p.endswith('.py') and not is_test_path(p))
        self.src = {p: blob(repo, tree[p][0]) for p in self.files}
        self.units = []  # (file, qualname, start, end, text)
        self.defs_by_name = defaultdict(set)  # short name -> files defining it
        self.file_defs = {}
        for p in self.files:
            s = self.src[p]
            lines = s.split('\n')
            spans = def_spans(s) or []
            self.file_defs[p] = spans
            covered = np.zeros(len(lines) + 2, dtype=bool)
            for q, k, a, b in spans:
                self.defs_by_name[q.split('.')[-1].lower()].add(p)
            # top-level-ish units: functions (any depth, innermost-free) and class headers
            for q, k, a, b in spans:
                if k == 'func':
                    # skip functions nested in functions (they're inside the parent's text)
                    parent = q.rsplit('.', 1)[0] if '.' in q else None
                    if parent and any(pq == parent and pk == 'func' for pq, pk, _, _ in spans):
                        continue
                    self.units.append((p, q, a, b, '\n'.join(lines[a - 1:b])))
                    covered[a:b + 1] = True
            rest = [l for i, l in enumerate(lines, 1) if not covered[i]]
            self.units.append((p, '<module>', 1, len(lines), '\n'.join(rest)))
        self.fidx = {p: i for i, p in enumerate(self.files)}
        # auxiliary (non-library) code: the same directories the harness prunes from its workspace layout
        self.aux = {p for p in self.files if p.split('/')[0].lower() in AUX_DIRS}
        self._file_bm25 = self._unit_bm25 = None

    # -- token costs (Gemma tokenizer)
    def file_tokens(self, p):
        return ntok(self.src[p])

    def unit_tokens(self, u):
        return ntok(u[4])

    def skeleton(self, p):
        """Signatures-only outline of a file (repo-map style)."""
        lines = self.src[p].split('\n')
        out = []
        for q, k, a, b in self.file_defs[p]:
            depth = q.count('.')
            sig = lines[a - 1].strip() if a - 1 < len(lines) else ''
            # skip decorators: find def/class line
            for j in range(a - 1, min(b, len(lines))):
                if re.match(r'\s*(async\s+def|def|class)\b', lines[j]):
                    sig = lines[j].strip(); break
            out.append('  ' * depth + sig)
        return p + ':\n' + '\n'.join(out)

    # -- BM25 indexes
    def file_bm25(self):
        if self._file_bm25 is None:
            docs = [text_tokens(p.replace('/', ' ') + ' ' + self.src[p]) for p in self.files]
            self._file_bm25 = BM25Okapi(docs)
        return self._file_bm25

    def unit_bm25(self):
        if self._unit_bm25 is None:
            docs = [text_tokens(u[0].replace('/', ' ') + ' ' + u[1] + ' ' + u[4]) for u in self.units]
            self._unit_bm25 = BM25Okapi(docs)
        return self._unit_bm25


# ---------------------------------------------------------------- history priors
@functools.lru_cache(maxsize=None)
def history(repo, commit, n=300):
    out = git(repo, 'log', '-n', str(n), '--name-only', '--format=@@%H', commit).decode()
    commits = []
    for block in out.split('@@')[1:]:
        lines = block.strip().split('\n')
        commits.append([l for l in lines[1:] if l.strip()])
    return commits


def recency_scores(repo, commit, files, n=300, half_life=25):
    commits = history(repo, commit, n)
    sc = defaultdict(float)
    for age, fl in enumerate(commits):
        w = 0.5 ** (age / half_life)
        for f in fl:
            sc[f] += w
    return np.array([sc.get(p, 0.0) for p in files])


# ---------------------------------------------------------------- static graph
def build_graph(idx):
    """File-level graph: import edges + name-reference edges (call by bare name of a def)."""
    mod_of = {}
    for p in idx.files:
        m = p[:-3].replace('/', '.')
        if m.endswith('.__init__'):
            m = m[:-9]
        mod_of[m] = p
    adj = defaultdict(set)
    for p in idx.files:
        try:
            tree = ast.parse(idx.src[p])
        except SyntaxError:
            continue
        pkg = p[:-3].replace('/', '.').rsplit('.', 1)[0]
        names = set()
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for a in node.names:
                    if a.name in mod_of: adj[p].add(mod_of[a.name])
            elif isinstance(node, ast.ImportFrom):
                base = node.module or ''
                if node.level:
                    parts = pkg.split('.')
                    up = parts[:len(parts) - (node.level - 1)] if node.level > 1 else parts
                    base = '.'.join(up + ([node.module] if node.module else []))
                if base in mod_of: adj[p].add(mod_of[base])
                for a in node.names:
                    full = f'{base}.{a.name}'
                    if full in mod_of: adj[p].add(mod_of[full])
            elif isinstance(node, ast.Call):
                f = node.func
                nm = f.id if isinstance(f, ast.Name) else (f.attr if isinstance(f, ast.Attribute) else None)
                if nm: names.add(nm.lower())
        for nm in names:
            defs = idx.defs_by_name.get(nm, ())
            if 0 < len(defs) <= 3:
                for q in defs:
                    if q != p: adj[p].add(q)
    und = defaultdict(set)
    for a, bs in adj.items():
        for b in bs:
            if a != b:
                und[a].add(b); und[b].add(a)
    return und


# ---------------------------------------------------------------- rankers (return file score vectors)
def rank_from_scores(files, scores):
    order = np.argsort(-scores, kind='stable')
    return [files[i] for i in order]


def s_bm25_file(idx, q):
    return np.asarray(idx.file_bm25().get_scores(q), dtype=float)


def s_bm25_unit(idx, q):
    us = np.asarray(idx.unit_bm25().get_scores(q), dtype=float)
    fs = np.zeros(len(idx.files))
    for u, s in zip(idx.units, us):
        i = idx.fidx[u[0]]
        fs[i] = max(fs[i], s)
    return fs, us


def s_ident(idx, idents):
    """Exact identifier hits: defs named like an identifier from the statement, and path matches."""
    fs = np.zeros(len(idx.files))
    n = len(idx.files)
    for ident in idents:
        low = ident.lower()
        hits = set(idx.defs_by_name.get(low, ()))
        for p in idx.files:
            stem = Path(p).stem.lower()
            if stem == low or low in p.lower().replace('/', '.').split('.'):
                hits.add(p)
        if hits:
            w = math.log(1 + n / len(hits))
            for p in hits:
                fs[idx.fidx[p]] += w
    return fs


def s_grep(idx, idents):
    """What `grep -rl` would give: files containing an identifier, IDF-weighted."""
    fs = np.zeros(len(idx.files))
    n = len(idx.files)
    for ident in idents:
        if len(ident) < 4: continue
        hits = [i for i, p in enumerate(idx.files) if ident in idx.src[p]]
        if hits and len(hits) < n * 0.5:
            w = math.log(1 + n / len(hits))
            for i in hits: fs[i] += w
    return fs


def rrf(*rankings, k=60, weights=None):
    weights = weights or [1.0] * len(rankings)
    sc = defaultdict(float)
    for w, r in zip(weights, rankings):
        for pos, f in enumerate(r):
            sc[f] += w / (k + pos + 1)
    return sorted(sc, key=lambda f: -sc[f])


def graph_rerank(idx, graph, base_scores, alpha=0.3, seeds=5):
    """One-hop propagation from top seeds: s' = s + alpha * max_{seed neighbor} s(seed)."""
    files = idx.files
    s = base_scores / (base_scores.max() + 1e-9)
    order = np.argsort(-s)[:seeds]
    boost = np.zeros(len(files))
    for i in order:
        for nb in graph.get(files[i], ()):
            j = idx.fidx.get(nb)
            if j is not None:
                boost[j] = max(boost[j], s[i])
    return s + alpha * boost

"""Run all localization methods on all tasks; write per-task results + aggregate metrics."""
import json, math, sys, time
from collections import defaultdict
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent))
from retrieval import *  # noqa

KS = [1, 3, 5, 10, 20]
BUDGETS = [1000, 2000, 4000, 8000, 16000, 32000]
CALLS = [1, 2, 3, 5, 10, 20]
READ_LINES = 150  # harness read_file cap


def gold_units(idx, gfile):
    """Map changed old lines to our unit inventory (top-level-or-method functions, else <module>)."""
    p = gfile['path']
    us = [u for u in idx.units if u[0] == p and u[1] != '<module>']
    out = set()
    for ln in gfile.get('changed_old_lines', []):
        best = None
        for u in us:
            if u[2] <= ln <= u[3] and (best is None or u[3] - u[2] < best[3] - best[2]):
                best = u
        out.add((p, best[1] if best else '<module>'))
    return out


def file_metrics(rank, gold):
    m = {}
    pos = [rank.index(g) + 1 if g in rank else 10 ** 6 for g in gold]
    for k in KS:
        m[f'hit@{k}'] = float(min(pos) <= k)
        m[f'recall@{k}'] = sum(p <= k for p in pos) / len(gold)
        m[f'all@{k}'] = float(max(pos) <= k)
    m['mrr'] = 1.0 / min(pos) if min(pos) < 10 ** 6 else 0.0
    m['first_rank'] = min(pos)
    return m


def budget_curve_files(idx, rank, gold, gunits):
    """Read whole files in rank order. Cost in tokens and in read_file calls (150 lines/call)."""
    tok = 0; calls = 0; res_tok = {}; res_calls = {}
    got_f = set(); got_u = set()
    events = []
    for f in rank[:60]:
        t = idx.file_tokens(f); c = math.ceil((idx.src[f].count('\n') + 1) / READ_LINES)
        tok += t; calls += c
        if f in gold: got_f.add(f); got_u |= {u for u in gunits if u[0] == f}
        events.append((tok, calls, len(got_f) / len(gold), len(got_u) / max(1, len(gunits))))
    for B in BUDGETS:
        ok = [e for e in events if e[0] <= B]
        res_tok[B] = (ok[-1][2], ok[-1][3]) if ok else (0.0, 0.0)
    for C in CALLS:
        ok = [e for e in events if e[1] <= C]
        res_calls[C] = (ok[-1][2], ok[-1][3]) if ok else (0.0, 0.0)
    return res_tok, res_calls


def budget_curve_units(idx, unit_order, gold, gunits, skeleton_files=()):
    """Read function-level units in rank order (optionally after paying for skeletons).
    Each unit read = one read_file call with a line range (units > 150 lines cost >1 call)."""
    tok = sum(ntok(idx.skeleton(f)) for f in skeleton_files); calls = 0
    got_f = set(); got_u = set(); events = [(tok, calls, 0.0, 0.0)]
    for ui in unit_order[:400]:
        u = idx.units[ui]
        tok += idx.unit_tokens(u); calls += max(1, math.ceil((u[4].count('\n') + 1) / READ_LINES))
        if (u[0], u[1]) in gunits:
            got_u.add((u[0], u[1])); got_f.add(u[0])
        events.append((tok, calls, len(got_f) / len(gold), len(got_u) / max(1, len(gunits))))
    res_tok, res_calls = {}, {}
    for B in BUDGETS:
        ok = [e for e in events if e[0] <= B]
        res_tok[B] = (ok[-1][2], ok[-1][3]) if ok else (0.0, 0.0)
    for C in CALLS:
        ok = [e for e in events if e[1] <= C]
        res_calls[C] = (ok[-1][2], ok[-1][3]) if ok else (0.0, 0.0)
    return res_tok, res_calls


def main():
    tasks = {json.loads(l)['instance_id']: json.loads(l) for l in open(ROOT / 'data' / 'tasks.jsonl')}
    gold_all = sorted((json.loads(l) for l in open(ROOT / 'data' / 'gold.jsonl')), key=lambda g: (g['repo'], g['base_commit']))
    out = open(ROOT / 'out' / 'per_task.jsonl', 'w')
    idx_cache = {}
    t0 = time.time()
    for n, g in enumerate(gold_all):
        t = tasks[g['instance_id']]
        key = (g['repo'], g['base_commit'])
        if key not in idx_cache:
            idx_cache.clear()
            idx_cache[key] = (RepoIndex(*key), None)
        idx, graph = idx_cache[key]
        if graph is None:

def ci(x):
    x = np.asarray(x, float)
    bs = rng.choice(x, (B, len(x))).mean(1)
    return float(x.mean()), float(np.percentile(bs, 2.5)), float(np.percentile(bs, 97.5))


def paired(a, b):
    """Paired bootstrap on mean difference a-b; two-sided p via sign of resampled diff; plus exact McNemar."""
    a, b = np.asarray(a, float), np.asarray(b, float)
    d = a - b
    idx = rng.integers(0, len(d), (B, len(d)))
    bs = d[idx].mean(1)
    p = 2 * min((bs <= 0).mean(), (bs >= 0).mean())
    n01 = int((a > b).sum()); n10 = int((a < b).sum())  # for binary metrics this is McNemar; otherwise a sign test
    n = n01 + n10
    mc = min(1.0, 2 * sum(math.comb(n, i) for i in range(0, min(n01, n10) + 1)) / 2 ** n) if n else 1.0
    return {'diff': float(d.mean()), 'lo': float(np.percentile(bs, 2.5)), 'hi': float(np.percentile(bs, 97.5)),
            'p_boot': float(min(1.0, max(p, 1 / B))), 'wins': n01, 'losses': n10, 'p_mcnemar': mc}

#!/usr/bin/env python3
"""loc32k - zero-dependency fault localizer for small-context coding agents.

Runs inside an offline sandbox with only the Python stdlib and git. Ranks the
repository's library files and functions for a task description, using
  * BM25 over function-level units (file score = best unit),
  * exact identifier hits (definitions / module names mentioned in the task),
  * grep-style substring hits, IDF-weighted,
  * git recency of the history *before* the current commit,
fused with reciprocal-rank fusion (k=60) plus a one-hop import/call-graph boost.

Usage
  python3 loc32k.py --root /workspace --query-file /tmp/task.txt            # ranked files + functions
  python3 loc32k.py --root /workspace --query-file /tmp/task.txt --show 3   # print code of unit #3
Output is kept under --max-chars (default 4800) to fit tool-output caps.
"""
import argparse, ast, math, os, re, subprocess, sys
from collections import defaultdict

AUX = {'docs', 'doc', 'docs_src', 'examples', 'example', 'benchmarks', 'benchmark', 'scripts', 'tools'}
STOP = set('''a an the of to in on for and or is are be by with as at from this that it its not no if else elif

CAMEL = re.compile(r'[A-Z]+(?=[A-Z][a-z])|[A-Z]?[a-z]+|[A-Z]+|\d+')
IDENT_PATTERNS = [r'`([^`\n]{2,80})`', r'\b([A-Za-z_]\w*(?:\.[A-Za-z_]\w*)+)\b', r'\b([A-Z][a-z0-9]+(?:[A-Z][a-z0-9]*)+)\b',
                  r'\b([a-z0-9]+(?:_[a-z0-9]+)+)\b', r'\b(\w+\.py)\b']


def is_test(p):
    parts = p.split('/'); n = parts[-1]
    return n.startswith('test_') or n.endswith('_test.py') or n == 'conftest.py' or any(x in ('tests', 'test', 'testing') for x in parts[:-1])


def toks(text):
    out = []
    for raw in re.findall(r'[A-Za-z_][A-Za-z0-9_]*', text):
        low = raw.lower()
        if ('_' in raw or re.search(r'[a-z][A-Z]', raw)) and low not in STOP:
            out.append(low)
        for p in re.split(r'[_\W]+', raw):
            for x in CAMEL.findall(p) if p else []:
                x = x.lower()
                if len(x) > 1 and x not in STOP:
                    out.append(x)
    return out


def idents(s):
    found = set()
    for pat in IDENT_PATTERNS:
        for m in re.findall(pat, s):
            for piece in re.findall(r'[A-Za-z_][A-Za-z0-9_]*', m):
                if len(piece) > 2 and piece.lower() not in STOP:
                    found.add(piece)
    return found


class BM25:  # same formulation as rank_bm25.BM25Okapi (k1=1.5, b=0.75, eps=0.25)
    def __init__(self, docs, k1=1.5, b=0.75, eps=0.25):
        self.k1, self.b = k1, b
        self.tf = []; df = defaultdict(int); self.dl = []
        for d in docs:
            c = defaultdict(int)
            for w in d: c[w] += 1
            self.tf.append(c); self.dl.append(len(d))
            for w in c: df[w] += 1
        n = len(docs); self.avg = sum(self.dl) / max(n, 1)
        self.idf = {}; neg = []; tot = 0.0
        for w, f in df.items():
            v = math.log(n - f + 0.5) - math.log(f + 0.5)
            self.idf[w] = v; tot += v
            if v < 0: neg.append(w)
        e = eps * tot / max(len(self.idf), 1)
        for w in neg: self.idf[w] = e

    def scores(self, q):
        out = [0.0] * len(self.tf)
        for w in q:
            idf = self.idf.get(w)
            if idf is None: continue
            for i, c in enumerate(self.tf):
                f = c.get(w)
                if f:
                    out[i] += idf * (f * (self.k1 + 1) / (f + self.k1 * (1 - self.b + self.b * self.dl[i] / self.avg)))
        return out


def spans(src):
    try:
        tree = ast.parse(src)
    except SyntaxError:
        return []
    res = []

    def visit(node, prefix):
        for ch in ast.iter_child_nodes(node):
            if isinstance(ch, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                q = f'{prefix}.{ch.name}' if prefix else ch.name
                s = min([ch.lineno] + [d.lineno for d in ch.decorator_list])
                res.append((q, 'class' if isinstance(ch, ast.ClassDef) else 'func', s, ch.end_lineno))
                visit(ch, q)
            else:
                visit(ch, prefix)
    visit(tree, '')
    return res


def load(root):
    out = subprocess.run(['git', '-C', root, 'ls-files', '*.py'], capture_output=True, text=True).stdout.split()
    if not out:
        out = [os.path.relpath(os.path.join(d, f), root) for d, _, fs in os.walk(root) for f in fs if f.endswith('.py')]
    files = sorted(p for p in out if not is_test(p) and '/.' not in '/' + p)
    src = {}
    for p in files:
        try:
            src[p] = open(os.path.join(root, p), encoding='utf-8', errors='replace').read()
        except OSError:
            pass
    return [p for p in files if p in src], src


def recency(root, files, n=300, hl=25):
    out = subprocess.run(['git', '-C', root, 'log', '-n', str(n), '--name-only', '--format=@@%H'], capture_output=True, text=True).stdout
    sc = defaultdict(float)
    # (the harness's single "baseline" commit on top only shifts every age by one: ranking unchanged)
    for age, block in enumerate(out.split('@@')[1:]):
        for f in [l for l in block.strip().split('\n')[1:] if l.strip()]:
            sc[f] += 0.5 ** (age / hl)
    return [sc.get(p, 0.0) for p in files]


def graph(files, src, defs_by_name):
    mod = {}
    for p in files:
        m = p[:-3].replace('/', '.'); m = m[:-9] if m.endswith('.__init__') else m
        mod[m] = p
    und = defaultdict(set)
    for p in files:
        try:
            tree = ast.parse(src[p])
        except SyntaxError:
            continue
        pkg = p[:-3].replace('/', '.').rsplit('.', 1)[0]; names = set(); nb = set()
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for a in node.names:
                    if a.name in mod: nb.add(mod[a.name])
            elif isinstance(node, ast.ImportFrom):
                base = node.module or ''
                if node.level:
                    parts = pkg.split('.')
                    up = parts[:len(parts) - (node.level - 1)] if node.level > 1 else parts
                    base = '.'.join(up + ([node.module] if node.module else []))
                if base in mod: nb.add(mod[base])
                for a in node.names:
                    if f'{base}.{a.name}' in mod: nb.add(mod[f'{base}.{a.name}'])
            elif isinstance(node, ast.Call):
                f = node.func
                nm = f.id if isinstance(f, ast.Name) else (f.attr if isinstance(f, ast.Attribute) else None)
                if nm: names.add(nm.lower())
        for nm in names:
            d = defs_by_name.get(nm, ())
            if 0 < len(d) <= 3: nb |= set(d)
        for q in nb:
            if q != p: und[p].add(q); und[q].add(p)
    return und


def rrf(rankings, k=60):
    sc = defaultdict(float)
    for r in rankings:
        for i, f in enumerate(r): sc[f] += 1.0 / (k + i + 1)
    return sorted(sc, key=lambda f: -sc[f])


def localize(root, query):
    files, src = load(root)
    fidx = {p: i for i, p in enumerate(files)}
    units = []; defs_by_name = defaultdict(set)
    for p in files:
        lines = src[p].split('\n'); sp = spans(src[p]); covered = [False] * (len(lines) + 2)
        funcs = {q for q, k, _, _ in sp if k == 'func'}
        for q, k, a, b in sp:
            defs_by_name[q.split('.')[-1].lower()].add(p)
        for q, k, a, b in sp:
            if k == 'func' and not ('.' in q and q.rsplit('.', 1)[0] in funcs):
                units.append((p, q, a, b, '\n'.join(lines[a - 1:b])))
                for i in range(a, b + 1): covered[i] = True
        units.append((p, '<module>', 1, len(lines), '\n'.join(l for i, l in enumerate(lines, 1) if not covered[i])))
    query = re.sub(r'https?://\S+', ' ', re.sub(r'<!--.*?-->', ' ', query, flags=re.S))
    q = toks(query)
    ids = idents(query)
    us = BM25([toks(u[0].replace('/', ' ') + ' ' + u[1] + ' ' + u[4]) for u in units]).scores(q) if q else [0.0] * len(units)
    fs = [0.0] * len(files)
    for u, s in zip(units, us): fs[fidx[u[0]]] = max(fs[fidx[u[0]]], s)
    n = len(files)
    ident = [0.0] * n; grep = [0.0] * n
    for w in ids:
        low = w.lower(); hits = set(defs_by_name.get(low, ()))
        hits |= {p for p in files if os.path.basename(p)[:-3].lower() == low or low in p.lower().replace('/', '.').split('.')}
        for p in hits: ident[fidx[p]] += math.log(1 + n / len(hits))
        if len(w) >= 4:
            gh = [i for i, p in enumerate(files) if w in src[p]]
            if gh and len(gh) < n * 0.5:
                for i in gh: grep[i] += math.log(1 + n / len(gh))
    rec = recency(root, files)

    def order(v, positive=False):
        o = sorted(range(n), key=lambda i: -v[i])
        return [files[i] for i in o if (v[i] > 0 or not positive)]
    bm = order(fs)
    fused = rrf([bm, order(ident, True), order(grep, True), order(rec, True)])
    fused += [f for f in bm if f not in set(fused)]
    s = [0.0] * n
    for i, f in enumerate(fused): s[fidx[f]] = 1.0 / (61 + i)
    mx = max(s) or 1.0; s = [x / mx for x in s]
    g = graph(files, src, defs_by_name)
    boost = [0.0] * n
    for i in sorted(range(n), key=lambda i: -s[i])[:5]:
        for nb in g.get(files[i], ()):
            j = fidx.get(nb)
            if j is not None: boost[j] = max(boost[j], s[i])
    final = [s[i] + 0.3 * boost[i] for i in range(n)]
    ranked = sorted(range(n), key=lambda i: -final[i])
    ranked_files = [files[i] for i in ranked]
    # two-stage unit ranking: units of the top-10 files by unit BM25, then the rest
    top10 = set(ranked_files[:10]); mu = max(us) if us and max(us) > 0 else 1.0
    uorder = sorted(range(len(units)), key=lambda i: -(us[i] / mu + (2.0 if units[i][0] in top10 else 0.0)))
    return ranked_files, [units[i] for i in uorder], src


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--root', default='/workspace'); ap.add_argument('--query'); ap.add_argument('--query-file')
    ap.add_argument('--k', type=int, default=10); ap.add_argument('--units', type=int, default=25)
    ap.add_argument('--show', type=int, default=None); ap.add_argument('--max-chars', type=int, default=4800)
    a = ap.parse_args()
    query = a.query or (open(a.query_file).read() if a.query_file else sys.stdin.read())
    files, units, src = localize(a.root, query)

        print(f'# {u[0]}::{u[1]} lines {u[2]}-{u[3]}\n' + body[:a.max_chars - 80]); return
